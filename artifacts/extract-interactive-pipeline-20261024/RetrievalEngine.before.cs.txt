internal static partial class RetrievalEngine
{

    private static async Task<RagSearchResponse> SearchInteractiveAsync(
        RetrievalExecutionContext context,
        NpgsqlDataSource ds,
        RagOptions rag,
        IHttpClientFactory httpFactory,
        RagSearchRequestDto req)
    {
        var tenantId = context.TenantId;
        var ct = context.CancellationToken;
        var teiGovernor = context.TeiGovernor;

        const string catalogOverviewFallbackQuery = "Which documents are available in this category?";

        var repairedQuery = RepairLikelyMojibakeText(req.Query);
        if (!string.Equals(repairedQuery, req.Query, StringComparison.Ordinal))
            req = req with { Query = repairedQuery };

        var topK = req.TopK ?? rag.DefaultTopK;
        topK = Math.Clamp(topK, 1, rag.MaxTopK);
        var requestedTopK = topK;

        var requestedCategory = string.IsNullOrWhiteSpace(req.Category)
            ? null
            : req.Category.Trim().ToLowerInvariant();
        var categoryPath = await ResolveRagCategoryPathAsync(ds, tenantId, req.CategoryPath, req.CategoryRef, ct);
        var category = string.IsNullOrWhiteSpace(categoryPath)
            ? requestedCategory
            : null;
        var categoryQueryHint = requestedCategory
                                ?? NormalizeRagCategory(ExtractTopLevelCategoryPath(categoryPath));

        var hasDocScope = !string.IsNullOrWhiteSpace(req.DocId) || !string.IsNullOrWhiteSpace(req.DocPath);
        var hasExplicitFileDocumentHint = HasExplicitFileLookupIntent(req.Query);
        var useBareVsComparativeOperandRoute = ShouldUseBareVsComparativeOperandRoute(req.Query);
        var useChoiceBetweenLexicalRoute = ShouldUseChoiceBetweenLexicalRoute(req.Query)
            && !hasDocScope
            && !hasExplicitFileDocumentHint;
        var retrievalQuery = useChoiceBetweenLexicalRoute
            ? BuildChoiceBetweenRetrievalQuery(req.Query)
            : ResolvePrimaryRetrievalQuery(req.Query, categoryQueryHint);
        retrievalQuery = ExpandGenericTechnicalRetrievalQuery(req.Query, retrievalQuery);
        var lexicalRetrievalSurface = ResolveLexicalRetrievalSurface(req.Query, retrievalQuery);
        var usesDelimitedLexicalSurface = !string.Equals(lexicalRetrievalSurface, req.Query, StringComparison.Ordinal);
        var queryNorm = NormalizeQuery(req.Query);
        var retrievalQueryNorm = NormalizeQuery(retrievalQuery);

        var includeResearchSurfaces = ShouldUseResearchSurfaces(req);
        var mode = ResolveEffectiveSearchMode(req.Mode, req.Query);
        if (includeResearchSurfaces && !string.Equals(mode, "focused", StringComparison.Ordinal))
            mode = "broad";
        topK = ResolveInternalSelectionTopK(
            req.Query,
            requestedTopK,
            rag.MaxTopK,
            hasDocScope,
            hasExplicitFileDocumentHint,
            mode);
        if (usesDelimitedLexicalSurface)
            topK = Math.Min(Math.Max(topK, 16), rag.MaxTopK);
        var preferItemInventory = ShouldPreferItemInventorySelection(req, mode, includeResearchSurfaces);
        var preferDocumentDiversity = (ShouldPreferDocumentDiversity(req.Query) || includeResearchSurfaces)
            && mode != "focused"
            && !preferItemInventory;
        double defMinScore = mode switch
        {
            "focused" => 0.35,
            "broad" => 0.15,
            _ => 0.25
        };
        int defCandidates = ResolveDefaultCandidateCount(mode, preferDocumentDiversity, topK);
        // CDC v3.1 §11.3: max 3 chunks per document default
        int defMaxPerDoc = ResolveDefaultMaxPerDoc(mode, preferDocumentDiversity, topK);

        var minScore = Math.Clamp(req.MinScore ?? defMinScore, 0.0, 1.0);
        var selectionMinScore = !req.MinScore.HasValue && ShouldRelaxDefaultScoreFloorForFocusedStructuredLookup(req.Query)
            ? 0.0
            : minScore;
        var candidates = Math.Clamp(req.Candidates ?? defCandidates, topK, Math.Max(topK, rag.MaxTopK * 20));
        var maxPerDoc = Math.Clamp(req.MaxPerDoc ?? (preferItemInventory ? topK : defMaxPerDoc), 1, topK);
        var maxPerPage = Math.Clamp(req.MaxPerPage ?? (preferItemInventory ? Math.Min(topK, 16) : 1), 1, topK);
        var hasExplicitMaxPerDoc = req.MaxPerDoc.HasValue;
        var hasExplicitMaxPerPage = req.MaxPerPage.HasValue;
        var requestedPageStart = hasDocScope && req.PageStart is > 0
            ? req.PageStart.Value
            : (int?)null;
        var requestedPageEnd = requestedPageStart.HasValue
            ? Math.Max(requestedPageStart.Value, req.PageEnd is > 0 ? req.PageEnd.Value : requestedPageStart.Value)
            : (int?)null;
        var hasRequestedPageWindow = requestedPageStart.HasValue && requestedPageEnd.HasValue;

        if (req.Diversity != null)
        {
            if (req.Diversity.MaxChunksPerDoc.HasValue)
            {
                maxPerDoc = Math.Clamp(req.Diversity.MaxChunksPerDoc.Value, 1, topK);
                hasExplicitMaxPerDoc = true;
            }
            if (req.Diversity.PreferDistinctPages == true)
            {
                maxPerPage = 1;
                hasExplicitMaxPerPage = true;
            }
        }

        if (!hasExplicitMaxPerDoc && ShouldConstrainPreciseTitleLookup(req.Query))
            maxPerDoc = Math.Min(topK, Math.Max(maxPerDoc, 2));
        if (!hasExplicitMaxPerDoc && ShouldAllowMultipleChunksForComparativeSubject(req.Query))
            maxPerDoc = Math.Min(topK, Math.Max(maxPerDoc, 2));
        if (!hasExplicitMaxPerPage && ShouldConstrainPreciseTitleLookup(req.Query))
            maxPerPage = Math.Min(topK, Math.Max(maxPerPage, 2));
        if (hasRequestedPageWindow && !hasExplicitMaxPerDoc)
            maxPerDoc = Math.Min(topK, Math.Max(maxPerDoc, Math.Min(topK, 4)));
        if (hasRequestedPageWindow && !hasExplicitMaxPerPage)
            maxPerPage = Math.Min(topK, Math.Max(maxPerPage, 2));

        var swTotal = Stopwatch.StartNew();
        var hasCategoryFilter = !string.IsNullOrWhiteSpace(category) || !string.IsNullOrWhiteSpace(categoryPath);
        var skipPreciseTitleLookupForDocumentOverview = ShouldSkipPreciseTitleLookupForDocumentOverview(
            req.Query,
            hasDocScope,
            mode);
        var skipChunkRetrieversForDocumentOverview = ShouldSkipChunkRetrieversForDocumentOverview(req.Query, hasDocScope, mode);
        var forceDocumentRoleProfileFallback = !hasDocScope
            && hasCategoryFilter
            && (ContainsDocumentRoleClassificationSurface(req.Query)
                || ContainsDelimitedDocumentRoleClassificationSurface(req.Query));
        var preferUnquotedTitleAnchorRoute = !hasDocScope
            && !forceDocumentRoleProfileFallback
            && ShouldProbeUnquotedTitleAnchorRoute(
                req.Query,
                skipChunkRetrieversForDocumentOverview,
                useScopedProfileFallback: false);
        var hasQuotedLookupPhrases = ExtractQuotedLookupPhrases(req.Query).Count > 0;
        var skipUnquotedPreciseTitleScans = preferUnquotedTitleAnchorRoute
            && !hasQuotedLookupPhrases;
        var skipUnquotedCategoryTitleScans = hasCategoryFilter
            && skipUnquotedPreciseTitleScans;
        var useScopedProfileFallback = !hasDocScope
            && (forceDocumentRoleProfileFallback
                || (!preferUnquotedTitleAnchorRoute
                    && ShouldUseScopedProfileFallback(req.Query, hasCategoryFilter, mode)));
        var useScopedCatalogDocumentOverviewFallback = !hasDocScope
            && hasCategoryFilter
            && ShouldUseScopedCatalogDocumentOverviewFallback(req.Query, mode);
        if (useScopedCatalogDocumentOverviewFallback)
        {
            retrievalQuery = catalogOverviewFallbackQuery;
            retrievalQueryNorm = NormalizeQuery(retrievalQuery);
        }
        var selectionQuery = useScopedCatalogDocumentOverviewFallback
            ? catalogOverviewFallbackQuery
            : usesDelimitedLexicalSurface
                ? lexicalRetrievalSurface
            : req.Query;
        var diagnostics = req.IncludeDiagnostics == true
            ? new RagRetrievalDiagnosticsBuilder(
                query: req.Query,
                retrievalQuery: retrievalQuery,
                selectionQuery: selectionQuery,
                mode: mode,
                category: requestedCategory ?? categoryQueryHint,
                categoryPath: categoryPath,
                topK: requestedTopK,
                candidates: candidates,
                maxPerDoc: maxPerDoc,
                maxPerPage: maxPerPage)
            : null;
        var allowSparseAssistForScopedProfileFallback =
            useScopedProfileFallback && ShouldAllowSparseAssistForScopedProfileFallback(req.Query);
        var allowSparseAssistForBroadDiversity =
            !hasDocScope && (includeResearchSurfaces || ShouldAllowSparseAssistForBroadDiversity(req.Query));
        var allowNavigationCatalogRouteForBroadExploration =
            includeResearchSurfaces
            || ShouldAllowNavigationCatalogRouteForBroadExploration(req.Query, hasCategoryFilter || hasDocScope);
        var skipUnanchoredTitleAnchorRouteForBroadDiversity =
            !includeResearchSurfaces
            && (useChoiceBetweenLexicalRoute
                || ShouldSkipUnanchoredTitleAnchorRouteForBroadDiversity(req.Query));
        var skipSparseRetrieverForBroadDiversity = !hasDocScope
            && !includeResearchSurfaces
            && !allowSparseAssistForBroadDiversity
            && (ShouldSkipSparseRetrieverForBroadDiversity(req.Query, mode)
                || ShouldSkipSparseRetrieverForQuantityLookup(req.Query, mode));
        var skipSparseProfileCardAssist = !hasDocScope
            && !includeResearchSurfaces
            && ShouldSkipSparseProfileCardAssist(req.Query, mode);
        var requireDocumentOverviewProfileMatch = skipChunkRetrieversForDocumentOverview
            && ShouldRequireDocumentOverviewProfileMatch(req.Query);
        using var searchActivity = RetrievalTelemetry.StartSearchActivity(mode, hasCategoryFilter, hasDocScope, topK, candidates, req.Query);
        var skipLocalTitleTokenRoute = forceDocumentRoleProfileFallback
            || skipUnquotedCategoryTitleScans
            || skipUnquotedPreciseTitleScans
            || ShouldSkipLocalTitleTokenRouteForIndexedTitleAnchors(
            req.Query,
            hasCategoryScope: hasCategoryFilter,
            hasDocScope: hasDocScope,
            skipChunkRetrieversForDocumentOverview: skipChunkRetrieversForDocumentOverview,
            useScopedProfileFallback: useScopedProfileFallback);
        var skipQuotedTitleFullScan = ShouldSkipQuotedTitleFullScan(req.Query, hasDocScope);

        var phases = new RetrievalPhaseDiagnostics(context.Logger);
        var degradation = new RetrievalDegradationTracker();
        var degradedRetrievers = degradation.Retrievers;
        var degradedRetrieverErrors = degradation.Errors;

        var (exactMatches, exactMs) = await phases.MeasureAsync(
            phaseName: "retrieval_exact_match",
            retriever: "exact_match",
            action: () => SearchSourceBackedCanonicalExactMatchesAsync(ds, tenantId, req.Query, category, req.DocId, req.DocPath, ResolveExactMatchCandidateLimit(topK), ct, categoryPath),
            getReturnedCount: static matches => matches.Count);
        long titleAnchorRoutePhaseMs = 0;
        var (quotedTitleMatches, quotedTitleMs) = await phases.MeasureAsync(
            phaseName: "retrieval_quoted_title",
            retriever: "quoted_title",
            action: () => skipPreciseTitleLookupForDocumentOverview || skipUnquotedPreciseTitleScans
                || skipQuotedTitleFullScan
                ? Task.FromResult(new List<RagMatch>())
                : SearchQuotedTitleMatchesAsync(ds, tenantId, req.Query, category, req.DocId, req.DocPath, topK, ct, categoryPath),
            getReturnedCount: static matches => matches.Count);
        var (quotedTitleAnchorMatches, quotedTitleAnchorMs) = await phases.MeasureAsync(
            phaseName: "retrieval_quoted_title_anchor_route",
            retriever: "title_anchor_route",
            action: async () =>
            {
                if (skipPreciseTitleLookupForDocumentOverview
                    || !hasQuotedLookupPhrases
                    || !hasCategoryFilter
                    || hasDocScope)
                {
                    return new List<RagMatch>();
                }

                var anchorMatches = new List<RagMatch>();
                foreach (var phrase in ExtractTitleLookupPhrases(req.Query).Take(3))
                {
                    var titleAnchorQuery = phrase.Contains('"', StringComparison.Ordinal)
                        ? phrase
                        : $"\"{phrase}\"";
                    anchorMatches.AddRange(await SearchTitleAnchorRouteMatchesAsync(
                        ds,
                        tenantId,
                        titleAnchorQuery,
                        category,
                        req.DocId,
                        req.DocPath,
                        Math.Min(candidates, Math.Max(topK, 16)),
                        ct,
                        categoryPath,
                        degradedRetrieverRef: degradation.Mark,
                        allowDirectChunkRoute: false));
                }

                return DeduplicatePreciseTitleBackfillMatches(anchorMatches)
                    .Take(Math.Min(candidates, Math.Max(topK, 16)))
                    .ToList();
            },
            getReturnedCount: static matches => matches.Count);
        titleAnchorRoutePhaseMs += quotedTitleAnchorMs;
        var (explicitDocumentTitleMatches, explicitDocumentTitleMs) = await phases.MeasureAsync(
            phaseName: "retrieval_explicit_document_title",
            retriever: "explicit_document_title_route",
            action: () => hasExplicitFileDocumentHint
                ? SearchExplicitDocumentTitleMatchesAsync(
                    ds,
                    tenantId,
                    req.Query,
                    category,
                    req.DocId,
                    req.DocPath,
                    Math.Min(candidates, Math.Max(topK, 16)),
                    ct,
                    categoryPath,
                    degradedRetrieverRef: degradation.Mark)
                : Task.FromResult(new List<RagMatch>()),
            getReturnedCount: static matches => matches.Count);
        titleAnchorRoutePhaseMs += explicitDocumentTitleMs;
        var (localTitleTokenMatches, localTitleTokenMs) = await phases.MeasureAsync(
            phaseName: "retrieval_local_title_token",
            retriever: "local_title_token_route",
            action: () => skipPreciseTitleLookupForDocumentOverview
                || skipLocalTitleTokenRoute
                ? Task.FromResult(new List<RagMatch>())
                : SearchLocalTitleTokenRouteMatchesAsync(
                ds,
                tenantId,
                req.Query,
                category,
                req.DocId,
                req.DocPath,
                ResolveLocalTitleTokenCandidateLimit(topK, hasDocScope),
                ct,
                categoryPath,
                degradedRetrieverRef: degradation.Mark),
            getReturnedCount: static matches => matches.Count);
        var titleLookupMatches = explicitDocumentTitleMatches
            .Concat(quotedTitleMatches)
            .Concat(quotedTitleAnchorMatches)
            .Concat(localTitleTokenMatches)
            .GroupBy(BuildMatchDedupKey, StringComparer.OrdinalIgnoreCase)
            .Select(static group => group
                .OrderByDescending(match => match.Score)
                .ThenBy(match => match.DocPath, StringComparer.OrdinalIgnoreCase)
                .ThenBy(match => match.ChunkIndex)
                .First())
            .OrderByDescending(static match => match.Score)
            .ThenBy(static match => match.DocPath, StringComparer.OrdinalIgnoreCase)
            .ThenBy(static match => match.ChunkIndex)
            .ToList();
        titleLookupMatches = PreferRequestedPageWindowMatches(titleLookupMatches, requestedPageStart, requestedPageEnd);
        diagnostics?.CapturePhase("exact_match", "exact_match", exactMs, exactMatches);
        diagnostics?.CapturePhase("quoted_title", "quoted_title", quotedTitleMs, quotedTitleMatches);
        diagnostics?.CapturePhase("quoted_title_anchor", "title_anchor_route", quotedTitleAnchorMs, quotedTitleAnchorMatches);
        diagnostics?.CapturePhase("explicit_document_title", "explicit_document_title_route", explicitDocumentTitleMs, explicitDocumentTitleMatches);
        diagnostics?.CapturePhase("local_title_token", "local_title_token_route", localTitleTokenMs, localTitleTokenMatches);
        diagnostics?.CapturePhase("title_lookup_combined", "title_lookup", null, titleLookupMatches);
        var explicitDocumentScopedBackfillQuery = hasExplicitFileDocumentHint
            ? BuildDocumentHintScopedBackfillQuery(req.Query, retrievalQuery)
            : string.Empty;
        var explicitDocumentTitleRouteReturnedContent = hasExplicitFileDocumentHint
            && explicitDocumentTitleMatches.Count > 0;
        var useExplicitDocumentScopedBackfill = ShouldUseExplicitDocumentScopedBackfillRoute(
            req.Query,
            explicitDocumentScopedBackfillQuery)
            && !explicitDocumentTitleRouteReturnedContent;
        var skipBroadRetrieversForExplicitFileHint = hasExplicitFileDocumentHint;
        var shortCircuitAfterExact = ShouldShortCircuitAfterExact(req.Query, exactMatches);
        var shortCircuitAfterQuotedTitle = false;
        var selected = new List<RagMatch>(capacity: topK);
        var selectedKeys = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
        List<RagMatch>? broadDiversityCandidateBackfillPool = null;

        long teiMs = 0;
        long rerankMs = 0;
        long sparseMs = 0;
        long qdrantMs = 0;
        long sparsePhaseMs = 0;
        long densePhaseMs = 0;
        long linkedPhaseMs = 0;
        long profilePhaseMs = 0;
        long fusionMs = 0;
        long rerankPhaseMs = 0;
        long selectionMs = 0;
        int qdrantStatus = 0;
        var sparseCommandTimeoutSeconds = RagOptions.ResolveSearchSparseCommandTimeoutSeconds(rag.SearchSparseCommandTimeoutSeconds);
        var sqlHeavyCandidateLimit = ResolveSqlHeavyRetrieverCandidateLimit(candidates, topK);

        Task<List<RagMatch>> SearchPreciseTitleRouteBackfillAsync(string titleBackfillQuery, int limit)
        {
            if (ShouldSkipLocalTitleTokenRouteForIndexedTitleAnchors(
                    titleBackfillQuery,
                    hasCategoryScope: hasCategoryFilter,
                    hasDocScope: hasDocScope,
                    skipChunkRetrieversForDocumentOverview: skipChunkRetrieversForDocumentOverview,
                    useScopedProfileFallback: useScopedProfileFallback))
            {
                return SearchTitleAnchorRouteMatchesAsync(
                    ds,
                    tenantId,
                    titleBackfillQuery,
                    category,
                    req.DocId,
                    req.DocPath,
                    limit,
                    ct,
                    categoryPath,
                    degradedRetrieverRef: degradation.Mark,
                    allowDirectChunkRoute: false);
            }

            return SearchLocalTitleTokenRouteMatchesAsync(
                ds,
                tenantId,
                titleBackfillQuery,
                category,
                req.DocId,
                req.DocPath,
                limit,
                ct,
                categoryPath,
                degradedRetrieverRef: degradation.Mark);
        }

        List<RagMatch>? earlyTitleAnchorRouteMatches = null;
        List<RagMatch> preciseTitleRouteReuseMatches = [];
        long earlyTitleAnchorRouteMs = 0;
        var shortCircuitAfterTitleAnchorRoute = false;
        if (!shortCircuitAfterExact
            && !shortCircuitAfterQuotedTitle
            && ShouldProbeUnquotedTitleAnchorRoute(req.Query, skipChunkRetrieversForDocumentOverview, useScopedProfileFallback))
        {
            var (probedTitleAnchorRouteMatches, measuredTitleAnchorRouteMs) = await phases.MeasureAsync(
                phaseName: "retrieval_title_anchor_route",
                retriever: "title_anchor_route",
                action: () => SearchTitleAnchorRouteMatchesAsync(
                    ds,
                    tenantId,
                    req.Query,
                    category,
                    req.DocId,
                    req.DocPath,
                    Math.Min(candidates, Math.Max(topK, 16)),
                    ct,
                    categoryPath,
                    degradedRetrieverRef: degradation.Mark,
                    allowNavigationCatalogRoute: allowNavigationCatalogRouteForBroadExploration),
                getReturnedCount: static matches => matches.Count);
            earlyTitleAnchorRouteMatches = probedTitleAnchorRouteMatches;
            earlyTitleAnchorRouteMs = measuredTitleAnchorRouteMs;
            shortCircuitAfterTitleAnchorRoute = ShouldShortCircuitAfterTitleAnchorRoute(req.Query, probedTitleAnchorRouteMatches);
        }

        if (shortCircuitAfterExact)
        {
            AddRankedMatches(selected, selectedKeys, exactMatches, topK, minScore: 0.0, maxPerDoc, maxPerPage);
        }
        else if (shortCircuitAfterQuotedTitle)
        {
            AddRankedMatches(
                selected,
                selectedKeys,
                exactMatches.Concat(titleLookupMatches),
                topK,
                minScore: 0.0,
                maxPerDoc,
                maxPerPage,
                query: req.Query);
        }
        else if (shortCircuitAfterTitleAnchorRoute && earlyTitleAnchorRouteMatches is not null)
        {
            titleAnchorRoutePhaseMs = earlyTitleAnchorRouteMs;
            preciseTitleRouteReuseMatches = BuildReusablePreciseTitleRouteMatches(earlyTitleAnchorRouteMatches);
            AddRankedMatches(
                selected,
                selectedKeys,
                exactMatches
                    .Concat(titleLookupMatches)
                    .Concat(RankTitleAnchorRouteShortCircuitMatches(earlyTitleAnchorRouteMatches)),
                topK,
                minScore: 0.0,
                maxPerDoc,
                maxPerPage,
                query: req.Query);
        }
        else
        {
            Task<(List<RagMatch> Result, long DurationMs)> titleAnchorRouteMatchesTask = earlyTitleAnchorRouteMatches is not null
                ? Task.FromResult((earlyTitleAnchorRouteMatches, earlyTitleAnchorRouteMs))
                : skipChunkRetrieversForDocumentOverview
                  || useScopedProfileFallback
                  || skipBroadRetrieversForExplicitFileHint
                  || (skipUnanchoredTitleAnchorRouteForBroadDiversity && !allowNavigationCatalogRouteForBroadExploration)
                ? Task.FromResult((new List<RagMatch>(), 0L))
                : phases.MeasureAsync(
                    phaseName: "retrieval_title_anchor_route",
                    retriever: "title_anchor_route",
                    action: () => SearchTitleAnchorRouteMatchesAsync(
                        ds,
                        tenantId,
                        req.Query,
                        category,
                        req.DocId,
                        req.DocPath,
                        Math.Min(candidates, Math.Max(topK, 16)),
                        ct,
                        categoryPath,
                        degradedRetrieverRef: degradation.Mark,
                        allowNavigationCatalogRoute: allowNavigationCatalogRouteForBroadExploration),
                    getReturnedCount: static matches => matches.Count);
            Task<(List<RagMatch> Result, long DurationMs)> sparseMatchesTask = skipChunkRetrieversForDocumentOverview
                || (useScopedProfileFallback && !allowSparseAssistForScopedProfileFallback)
                || (skipSparseRetrieverForBroadDiversity && !allowSparseAssistForScopedProfileFallback)
                || skipBroadRetrieversForExplicitFileHint
                || useChoiceBetweenLexicalRoute
                || useBareVsComparativeOperandRoute
                ? Task.FromResult((new List<RagMatch>(), 0L))
                : phases.MeasureAsync(
                    phaseName: "retrieval_sparse",
                    retriever: "sparse_bm25",
                    action: () => SearchSparseMatchesAsync(
                        ds,
                        tenantId,
                        lexicalRetrievalSurface,
                        category,
                        req.DocId,
                        req.DocPath,
                        sqlHeavyCandidateLimit,
                        ct,
                        sparseMsRef: value => sparseMs = value,
                        lexicalExpansionQuery: retrievalQuery,
                        categoryPath: categoryPath,
                        includeProfileCardMatches: !skipSparseProfileCardAssist && !usesDelimitedLexicalSurface,
                        degradedRetrieverRef: degradation.Mark,
                        commandTimeoutSeconds: sparseCommandTimeoutSeconds),
                    getReturnedCount: static matches => matches.Count);
            Task<(List<RagMatch> Result, long DurationMs)> denseMatchesTask = skipChunkRetrieversForDocumentOverview
                || useScopedProfileFallback
                || skipBroadRetrieversForExplicitFileHint
                ? Task.FromResult((new List<RagMatch>(), 0L))
                : phases.MeasureAsync(
                    phaseName: "retrieval_dense",
                    retriever: "dense_qdrant",
                    action: () => SearchDenseMatchesAsync(
                        ds,
                        httpFactory,
                        rag,
                        tenantId,
                        retrievalQueryNorm,
                        category,
                        req.DocId,
                        req.DocPath,
                        candidates,
                        ct,
                        teiMsRef: value => teiMs = value,
                        qdrantMsRef: value => qdrantMs = value,
                        qdrantStatusRef: value => qdrantStatus = value,
                        categoryPath: categoryPath,
                        degradedRetrieverRef: degradation.Mark,
                        teiGovernor: teiGovernor),
                    getReturnedCount: static matches => matches.Count);
            var skipDocumentProfileSearchForPreciseLookup =
                !includeResearchSurfaces
                &&
                !skipChunkRetrieversForDocumentOverview
                && !useScopedProfileFallback
                && !useChoiceBetweenLexicalRoute
                && (useBareVsComparativeOperandRoute
                    || ShouldSkipDocumentProfileSearchForPreciseLookup(req.Query)
                    || ShouldSkipDocumentProfileSearchForComparativeLookup(req.Query));
            var allowComparativeDocumentProfileAssist = ShouldAllowDocumentProfileAssistForComparativeLookup(req.Query);
            var allowResearchDocumentProfileAssist = includeResearchSurfaces || allowComparativeDocumentProfileAssist;
            var documentProfileCandidateCount = ResolveDocumentProfileCandidateCount(
                candidates,
                topK,
                skipChunkRetrieversForDocumentOverview || useScopedProfileFallback,
                allowResearchDocumentProfileAssist);
            var canSearchDocumentProfiles = includeResearchSurfaces
                || (!skipDocumentProfileSearchForPreciseLookup
                && (allowComparativeDocumentProfileAssist || !ShouldSkipDocumentProfileSearchForQuantityLookup(req.Query))
                && (skipChunkRetrieversForDocumentOverview
                    || useScopedProfileFallback
                    || useChoiceBetweenLexicalRoute
                    || !ShouldSkipDocumentProfileSearchForLowCostBroadQuery(req.Query, mode)));
            var deferDocumentProfileSearch = !includeResearchSurfaces && !allowComparativeDocumentProfileAssist && ShouldDeferDocumentProfileSearch(
                canSearchDocumentProfiles,
                skipChunkRetrieversForDocumentOverview,
                useScopedProfileFallback,
                allowSparseAssistForScopedProfileFallback);
            Task<(List<RagMatch> Result, long DurationMs)> profileMatchesTask = !canSearchDocumentProfiles || deferDocumentProfileSearch
                || skipBroadRetrieversForExplicitFileHint
                ? Task.FromResult((new List<RagMatch>(), 0L))
                : SearchDocumentProfilesMeasuredAsync();
            Task<(List<RagMatch> Result, long DurationMs)> explicitDocumentScopedMatchesTask = !useExplicitDocumentScopedBackfill
                ? Task.FromResult((new List<RagMatch>(), 0L))
                : phases.MeasureAsync(
                    phaseName: "retrieval_explicit_document_scoped_backfill",
                    retriever: "document_hint_scoped_backfill",
                    action: () => SearchDocumentHintScopedBackfillMatchesAsync(
                        ds,
                        tenantId,
                        req.Query,
                        explicitDocumentScopedBackfillQuery,
                        category,
                        categoryPath,
                        Math.Max(topK, 8),
                        ct,
                        degradedRetrieverRef: degradation.Mark,
                        includeProfileCardMatches: false,
                        commandTimeoutSeconds: sparseCommandTimeoutSeconds),
                    getReturnedCount: static matches => matches.Count);

            Task<(List<RagMatch> Result, long DurationMs)> SearchDocumentProfilesMeasuredAsync()

                    lateStandardReferencePhrases,
                    lateStandardReferenceMatches)
                    ? lateStandardReferenceMatches
                    : lateStandardReferenceMatches.Concat(selected).ToList();
                var mergedStandardReferenceSelections = CalibrateFusedMatches(
                    req.Query,
                    lateStandardReferenceSelectionPool,
                    req.Query);
                selected.Clear();
                selectedKeys.Clear();
                AddRankedMatches(
                    selected,
                    selectedKeys,
                    mergedStandardReferenceSelections,
                    topK,
                    minScore: 0.0,
                    Math.Max(maxPerDoc, 3),
                    Math.Max(maxPerPage, 2),
                    query: req.Query);
            }
        }

        finalOrderingSw = Stopwatch.StartNew();
        CleanupLateSelectionBackfills(req.Query, selected);
        PruneUnrelatedReferenceVersionGroupSelections(req.Query, selected);
        PruneCurrentReferenceVersionSelectionsWhenHistoricalVersionRequested(req.Query, selected);
        PruneUnrequestedReferenceVersionYearSelections(req.Query, selected);
        PruneSupersededReferenceVersionSelections(req.Query, selected);
        PrioritizeFinalSelections(req.Query, selected);
        PrioritizeResponseDiversityCoverage(req.Query, selected, requestedTopK);
        ReattachStructuredContinuationCompanions(req.Query, selected, topK);
        PruneQualifiedQuotedTitleCompetitorDocuments(req.Query, selected);
        PrunePreciseTitleLocalNeighborhoodSelections(req.Query, selected);
        RebuildSelectedKeys(selected, selectedKeys);
        ReattachDocumentStatusOperandCompanions(
            req.Query,
            documentStatusOperandTerms,
            documentStatusOperandSelectionPool,
            selected,
            selectedKeys,
            topK);
        RebuildSelectedKeys(selected, selectedKeys);
        finalOrderingSw.Stop();
        selectionMs += finalOrderingSw.ElapsedMilliseconds;

        if (dateDocumentBackfillPhrases.Count > 0 && !SelectionsCoverDateAnchoredDocument(req.Query, selected))
        {
            var (lateDateDocumentBackfillMatches, lateDateDocumentBackfillMs) = await phases.MeasureAsync(
                phaseName: "retrieval_date_document_name_late_recovery",
                retriever: "date_document_name",
                action: () => SearchDateAnchoredDocumentNameContentMatchesAsync(
                    ds,
                    tenantId,
                    req.Query,
                    dateDocumentBackfillPhrases,
                    category,
                    req.DocId,
                    req.DocPath,
                    Math.Min(candidates, Math.Max(topK, 16)),
                    ct,
                    categoryPath,
                    degradedRetrieverRef: degradation.Mark),
                getReturnedCount: static matches => matches.Count);
            titleAnchorRoutePhaseMs += lateDateDocumentBackfillMs;

            if (lateDateDocumentBackfillMatches.Count > 0)
            {
                finalOrderingSw.Restart();
                selected.Clear();
                selectedKeys.Clear();
                var calibratedLateDateDocumentBackfill = SuppressNavigationalNoise(
                    req.Query,
                    CalibrateFusedMatches(req.Query, lateDateDocumentBackfillMatches, req.Query));
                AddRankedMatches(
                    selected,
                    selectedKeys,
                    calibratedLateDateDocumentBackfill,
                    topK,
                    minScore: 0.0,
                    maxPerDoc: Math.Max(maxPerDoc, 3),
                    maxPerPage: Math.Max(maxPerPage, 2),
                    query: req.Query);
                PrioritizeFinalSelections(req.Query, selected);
                RebuildSelectedKeys(selected, selectedKeys);
                finalOrderingSw.Stop();
                selectionMs += finalOrderingSw.ElapsedMilliseconds;
            }
        }

        if (ShouldUseReferenceVersionOperandRoute(req.Query)
            && !SelectionsCoverReferenceVersionOperandContent(req.Query, selected))
        {
            var lateReferenceVersionGroupKeys = ExtractReferenceVersionGroupKeysFromQuery(req.Query).ToList();
            if (lateReferenceVersionGroupKeys.Count == 0)
                lateReferenceVersionGroupKeys.AddRange(InferReferenceVersionOperandGroupKeys(req.Query, selected));
            lateReferenceVersionGroupKeys = lateReferenceVersionGroupKeys
                .Where(static key => !string.IsNullOrWhiteSpace(key))
                .Distinct(StringComparer.Ordinal)
                .Take(6)
                .ToList();

            if (lateReferenceVersionGroupKeys.Count > 0)
            {
                var (lateReferenceVersionOperandMatches, lateReferenceVersionOperandMs) = await phases.MeasureAsync(
                    phaseName: "retrieval_reference_version_operand_late_recovery",
                    retriever: "reference_version_operand_document_name",
                    action: () => SearchReferenceVersionOperandDocumentNameContentMatchesAsync(
                        ds,
                        tenantId,
                        req.Query,
                        lateReferenceVersionGroupKeys,
                        category,
                        req.DocId,
                        req.DocPath,
                        Math.Min(candidates, Math.Max(topK, 16)),
                        ct,
                        categoryPath,
                        degradedRetrieverRef: degradation.Mark),
                    getReturnedCount: static matches => matches.Count);
                titleAnchorRoutePhaseMs += lateReferenceVersionOperandMs;

                if (lateReferenceVersionOperandMatches.Count > 0)
                {
                    finalOrderingSw.Restart();
                    RebuildSelectedKeys(selected, selectedKeys);
                    var calibratedLateReferenceVersionOperand = SuppressNavigationalNoise(
                        req.Query,
                        CalibrateFusedMatches(req.Query, lateReferenceVersionOperandMatches, req.Query));
                    AddRankedMatches(
                        selected,
                        selectedKeys,
                        calibratedLateReferenceVersionOperand,
                        topK,
                        minScore: 0.0,
                        maxPerDoc: Math.Max(maxPerDoc, 2),
                        maxPerPage: Math.Max(maxPerPage, 2),
                        query: req.Query);
                    EnsureComparativeBackfillDocumentDiversity(
                        selected,
                        selectedKeys,
                        calibratedLateReferenceVersionOperand,
                        topK,
                        minScore: 0.0,
                        maxPerPage: Math.Max(maxPerPage, 2),
                        query: req.Query);
                    PruneUnrelatedReferenceVersionGroupSelections(req.Query, selected);
                    PruneCurrentReferenceVersionSelectionsWhenHistoricalVersionRequested(req.Query, selected);
                    PrioritizeFinalSelections(req.Query, selected);
                    RebuildSelectedKeys(selected, selectedKeys);
                    finalOrderingSw.Stop();
                    selectionMs += finalOrderingSw.ElapsedMilliseconds;
                }
            }
        }

        if (selected.Count == 0 && !hasExplicitFileDocumentHint)
        {
            var lateDocumentStatusOperandTerms = ExtractDocumentStatusOperandLookupTerms(req.Query);
            if (lateDocumentStatusOperandTerms.Count > 0)
            {
                var (lateDocumentStatusRecoveryMatches, lateDocumentStatusRecoveryMs) = await phases.MeasureAsync(
                    phaseName: "retrieval_document_status_operand_empty_recovery",
                    retriever: "document_status_operand",
                    action: () => SearchDocumentStatusOperandContentMatchesAsync(
                        ds,
                        tenantId,
                        req.Query,
                        lateDocumentStatusOperandTerms,
                        category,
                        req.DocId,
                        req.DocPath,
                        Math.Min(candidates, Math.Max(topK, 16)),
                        ct,
                        categoryPath,
                        degradedRetrieverRef: degradation.Mark),
                    getReturnedCount: static matches => matches.Count);
                titleAnchorRoutePhaseMs += lateDocumentStatusRecoveryMs;

                if (lateDocumentStatusRecoveryMatches.Count > 0)
                {
                    finalOrderingSw.Restart();
                    var calibratedLateDocumentStatusRecovery = SuppressNavigationalNoise(
                        req.Query,
                        CalibrateFusedMatches(req.Query, lateDocumentStatusRecoveryMatches, req.Query));
                    AddRankedMatches(
                        selected,
                        selectedKeys,
                        calibratedLateDocumentStatusRecovery,
                        topK,
                        minScore: 0.0,
                        maxPerDoc: Math.Max(maxPerDoc, 2),
                        maxPerPage: Math.Max(maxPerPage, 2),
                        query: req.Query);
                    PrioritizeFinalSelections(req.Query, selected);
                    RebuildSelectedKeys(selected, selectedKeys);
                    finalOrderingSw.Stop();
                    selectionMs += finalOrderingSw.ElapsedMilliseconds;
                }
            }
        }

        if (selected.Count == 0
            && hasCategoryFilter
            && ShouldUseGenericVersionConflictOverviewRecovery(req.Query))
        {
            var (genericVersionOverviewMatches, genericVersionOverviewMs) = await phases.MeasureAsync(
                phaseName: "retrieval_generic_version_conflict_overview_recovery",
                retriever: "document_profile",
                action: () => SearchDocumentOverviewProfileMatchesAsync(
                    ds,
                    tenantId,
                    catalogOverviewFallbackQuery,
                    category,
                    req.DocId,
                    req.DocPath,
                    Math.Min(Math.Max(topK * 2, 12), 24),
                    ct,
                    categoryPath,
                    requireLexicalMatch: false,
                    degradedRetrieverRef: degradation.Mark,
                    commandTimeoutSeconds: sparseCommandTimeoutSeconds),
                getReturnedCount: static matches => matches.Count);
            profilePhaseMs += genericVersionOverviewMs;

            if (genericVersionOverviewMatches.Count > 0)
            {
                finalOrderingSw.Restart();
                AddRankedMatches(
                    selected,
                    selectedKeys,
                    CalibrateFusedMatches(catalogOverviewFallbackQuery, genericVersionOverviewMatches, req.Query),
                    topK,
                    minScore: 0.0,
                    maxPerDoc: 1,
                    maxPerPage: Math.Max(maxPerPage, 1),
                    query: req.Query);
                RebuildSelectedKeys(selected, selectedKeys);
                finalOrderingSw.Stop();
                selectionMs += finalOrderingSw.ElapsedMilliseconds;
            }
        }

        if (hasExplicitFileDocumentHint
            && !SelectionsCoverExplicitFileLookupGroups(req.Query, selected))
        {
            var (lateExplicitFileProfileMatches, lateExplicitFileProfileMs) = await phases.MeasureAsync(
                phaseName: "retrieval_explicit_file_profile_late_recovery",
                retriever: "document_hint_profile_recovery",
                action: () => SearchExplicitFileHintProfileRecoveryMatchesAsync(
                    ds,
                    tenantId,
                    req.Query,
                    category,
                    categoryPath,
                    Math.Min(Math.Max(topK, 8), 16),
                    ct,
                    degradedRetrieverRef: degradation.Mark,
                    commandTimeoutSeconds: sparseCommandTimeoutSeconds),
                getReturnedCount: static matches => matches.Count);
            profilePhaseMs += lateExplicitFileProfileMs;

            if (lateExplicitFileProfileMatches.Count > 0)
            {
                finalOrderingSw.Restart();
                selected.Clear();
                selectedKeys.Clear();
                AddRankedMatches(
                    selected,
                    selectedKeys,
                    CalibrateFusedMatches(req.Query, lateExplicitFileProfileMatches, req.Query),
                    topK,
                    minScore: 0.0,
                    maxPerDoc: Math.Max(maxPerDoc, 2),
                    maxPerPage: Math.Max(maxPerPage, 1),
                    query: req.Query);
                RebuildSelectedKeys(selected, selectedKeys);
                finalOrderingSw.Stop();
                selectionMs += finalOrderingSw.ElapsedMilliseconds;
            }
        }

        var finalStandardReferencePhrases = ExtractStandardReferenceTitleLookupPhrases(req.Query);
        if (finalStandardReferencePhrases.Count > 1
            && !SelectionsCoverStandardReferencePhrases(finalStandardReferencePhrases, selected))
        {
            var (finalStandardReferenceProfileMatches, finalStandardReferenceProfileMs) = await phases.MeasureAsync(
                phaseName: "retrieval_standard_reference_profile_final_recovery",
                retriever: "standard_reference_document_name_profile",
                action: () => SearchStandardReferenceDocumentNameProfileMatchesAsync(
                    ds,
                    tenantId,
                    req.Query,
                    finalStandardReferencePhrases,
                    category,
                    req.DocId,
                    req.DocPath,
                    Math.Min(candidates, Math.Max(topK, 16)),
                    ct,
                    categoryPath,
                    degradedRetrieverRef: degradation.Mark,
                    commandTimeoutSeconds: sparseCommandTimeoutSeconds),
                getReturnedCount: static matches => matches.Count);
            profilePhaseMs += finalStandardReferenceProfileMs;

            if (SelectionsCoverStandardReferencePhrases(finalStandardReferencePhrases, finalStandardReferenceProfileMatches))
            {
                finalOrderingSw.Restart();
                selected.Clear();
                selectedKeys.Clear();
                AddRankedMatches(
                    selected,
                    selectedKeys,
                    CalibrateFusedMatches(req.Query, finalStandardReferenceProfileMatches, req.Query),
                    topK,
                    minScore: 0.0,
                    maxPerDoc: Math.Max(maxPerDoc, 2),
                    maxPerPage: Math.Max(maxPerPage, 1),
                    query: req.Query);
                PrioritizeFinalSelections(req.Query, selected);
                RebuildSelectedKeys(selected, selectedKeys);
                finalOrderingSw.Stop();
                selectionMs += finalOrderingSw.ElapsedMilliseconds;
            }
        }

        PruneUnrequestedStandardReferenceSelections(req.Query, selected);
        RebuildSelectedKeys(selected, selectedKeys);

        PrioritizeShortTechnicalDirectEvidenceSelections(req.Query, selected);
        RebuildSelectedKeys(selected, selectedKeys);

        if (ShouldSuppressUnscopedAmbiguousDocumentReferenceSources(req.Query, req.DocId, req.DocPath))
        {
            selected.Clear();
            selectedKeys.Clear();
        }

        if (selected.Count == 0
            && !skipChunkRetrieversForDocumentOverview
            && ShouldAllowCompactDocumentCodeEmptyRecovery(req.Query, hasExplicitFileDocumentHint)
            && compactDocumentCodeLookupPhrases.Count > 0)
        {
            var finalCompactCodeRecoveryLimit = Math.Min(candidates, Math.Max(topK * 4, 24));
            var (finalCompactCodeRecoveryMatches, finalCompactCodeRecoveryMs) = await phases.MeasureAsync(
                phaseName: "retrieval_compact_document_code_final_empty_recovery",
                retriever: "compact_document_code_recovery",
                action: async () =>
                {
                    var recovery = new List<RagMatch>();
                    foreach (var compactCode in compactDocumentCodeLookupPhrases.Take(3))
                    {
                        recovery.AddRange(await SearchTitleAnchorRouteMatchesAsync(
                            ds,
                            tenantId,
                            compactCode,
                            category,
                            req.DocId,
                            req.DocPath,
                            finalCompactCodeRecoveryLimit,
                            ct,
                            categoryPath,
                            degradedRetrieverRef: degradation.Mark));
                        recovery.AddRange(await SearchDenseMatchesAsync(
                            ds,
                            httpFactory,
                            rag,
                            tenantId,
                            NormalizeQuery(compactCode),
                            category,
                            req.DocId,
                            req.DocPath,
                            finalCompactCodeRecoveryLimit,
                            ct,
                            teiMsRef: value => teiMs += value,
                            qdrantMsRef: value => qdrantMs += value,
                            qdrantStatusRef: value => qdrantStatus = value,
                            categoryPath: categoryPath,
                            degradedRetrieverRef: degradation.Mark,
                            teiGovernor: teiGovernor));
                    }

                    return recovery
                        .GroupBy(BuildMatchDedupKey, StringComparer.OrdinalIgnoreCase)
                        .Select(static group => group
                            .OrderByDescending(match => match.Score)
                            .ThenBy(match => match.DocPath, StringComparer.OrdinalIgnoreCase)
                            .ThenBy(match => match.ChunkIndex)
                            .First())
                        .ToList();
                },
                getReturnedCount: static matches => matches.Count);
            titleAnchorRoutePhaseMs += finalCompactCodeRecoveryMs;

            if (finalCompactCodeRecoveryMatches.Count > 0)
            {
                finalOrderingSw.Restart();
                var compactRecoveryQuery = string.Join(' ', compactDocumentCodeLookupPhrases);
                var calibratedRecovery = CalibrateFusedMatches(
                    compactRecoveryQuery,
                    finalCompactCodeRecoveryMatches,
                    req.Query);
                AddRankedMatches(
                    selected,
                    selectedKeys,
                    SuppressNavigationalNoise(req.Query, calibratedRecovery),
                    topK,
                    minScore: 0.0,
                    Math.Max(maxPerDoc, 2),
                    Math.Max(maxPerPage, 2),
                    query: compactRecoveryQuery);
                PruneNavigationalSelections(req.Query, selected);
                RebuildSelectedKeys(selected, selectedKeys);
                finalOrderingSw.Stop();
                selectionMs += finalOrderingSw.ElapsedMilliseconds;
            }
        }

        var finalCompactDocumentCodeProfilePhrases = ExtractCompactDocumentCodeProfileRecoveryPhrases(req.Query)
            .Where(phrase => !SelectionsCoverCompactDocumentCodePhrase(phrase, selected))
            .ToArray();
        if (finalCompactDocumentCodeProfilePhrases.Length > 0
            && !skipChunkRetrieversForDocumentOverview
            && ShouldAllowCompactDocumentCodeEmptyRecovery(req.Query, hasExplicitFileDocumentHint)
            && (selected.Count == 0 || ShouldPreferComparativeDocumentDiversity(req.Query)))
        {
            var (finalCompactProfileMatches, finalCompactProfileMs) = await phases.MeasureAsync(
                phaseName: "retrieval_compact_document_code_profile_final_recovery",
                retriever: "compact_document_code_profile",
                action: async () =>
                {
                    var backfill = new List<RagMatch>();
                    foreach (var phrase in finalCompactDocumentCodeProfilePhrases.Take(4))
                    {
                        backfill.AddRange(await SearchDocumentOverviewProfileMatchesAsync(
                            ds,
                            tenantId,
                            phrase,
                            category,
                            req.DocId,
                            req.DocPath,
                            Math.Min(Math.Max(topK, 8), 16),
                            ct,
                            categoryPath,
                            requireLexicalMatch: true,
                            degradedRetrieverRef: degradation.Mark,
                            commandTimeoutSeconds: sparseCommandTimeoutSeconds));
                    }

                    return DeduplicatePreciseTitleBackfillMatches(backfill);
                },
                getReturnedCount: static matches => matches.Count);
            profilePhaseMs += finalCompactProfileMs;

            if (finalCompactProfileMatches.Count > 0)
            {
                finalOrderingSw.Restart();
                var compactProfileQuery = string.Join(' ', finalCompactDocumentCodeProfilePhrases);
                AddRankedMatches(
                    selected,
                    selectedKeys,
                    CalibrateFusedMatches(compactProfileQuery, finalCompactProfileMatches, req.Query),
                    topK,
                    minScore: 0.0,
                    Math.Max(maxPerDoc, 2),
                    Math.Max(maxPerPage, 1),
                    query: compactProfileQuery);
                PrioritizeFinalSelections(req.Query, selected);
                RebuildSelectedKeys(selected, selectedKeys);
                finalOrderingSw.Stop();
                selectionMs += finalOrderingSw.ElapsedMilliseconds;
            }
        }

        if (selected.Count == 0
            && ShouldAllowLexicalProfileEmptyRecovery(req.Query, hasExplicitFileDocumentHint))
        {
            var lexicalProfileRecoveryQuery = BuildLexicalProfileEmptyRecoveryQuery(req.Query, categoryQueryHint);
            if (!string.IsNullOrWhiteSpace(lexicalProfileRecoveryQuery))
            {
                var (lexicalProfileRecoveryMatches, lexicalProfileRecoveryMs) = await phases.MeasureAsync(
                    phaseName: "retrieval_lexical_profile_empty_recovery",
                    retriever: "lexical_profile_recovery",
                    action: () => SearchDocumentOverviewProfileMatchesAsync(
                        ds,
                        tenantId,
                        lexicalProfileRecoveryQuery,
                        category,
                        req.DocId,
                        req.DocPath,
                        Math.Min(Math.Max(topK, 8), 16),
                        ct,
                        categoryPath,
                        requireLexicalMatch: true,
                        degradedRetrieverRef: degradation.Mark,
                        commandTimeoutSeconds: sparseCommandTimeoutSeconds),
                    getReturnedCount: static matches => matches.Count);
                profilePhaseMs += lexicalProfileRecoveryMs;

                if (lexicalProfileRecoveryMatches.Count > 0)
                {
                    finalOrderingSw.Restart();
                    AddRankedMatches(
                        selected,
                        selectedKeys,
                        CalibrateFusedMatches(lexicalProfileRecoveryQuery, lexicalProfileRecoveryMatches, req.Query),
                        topK,
                        minScore: 0.0,
                        Math.Max(maxPerDoc, 2),
                        Math.Max(maxPerPage, 1),
                        query: lexicalProfileRecoveryQuery);
                    PrioritizeFinalSelections(req.Query, selected);
                    RebuildSelectedKeys(selected, selectedKeys);
                    finalOrderingSw.Stop();
                    selectionMs += finalOrderingSw.ElapsedMilliseconds;
                }
            }
        }

        if (hasRequestedPageWindow && selected.Count > 0)
        {
            finalOrderingSw.Restart();
            var preferredSelections = PreferRequestedPageWindowMatches(selected, requestedPageStart, requestedPageEnd);
            selected.Clear();
            selected.AddRange(preferredSelections.Take(topK));
            RebuildSelectedKeys(selected, selectedKeys);
            finalOrderingSw.Stop();
            selectionMs += finalOrderingSw.ElapsedMilliseconds;
        }

        swTotal.Stop();
        diagnostics?.CapturePhase("final_selected", "final_selection", selectionMs, selected);

        var response = new RagSearchResponse(
            RequestId: context.RequestId,
            Query: req.Query,
            QueryNormalized: queryNorm,
            Category: requestedCategory ?? categoryQueryHint,
            TopK: requestedTopK,
            MinScore: selectionMinScore,
            Candidates: candidates,
            MaxPerDoc: maxPerDoc,
            MaxPerPage: maxPerPage,
            QdrantStatus: qdrantStatus,
            Timings: new RagSearchTimings(
                TotalMs: swTotal.ElapsedMilliseconds,
                TeiMs: teiMs,
                RerankMs: rerankMs,
                SparseMs: sparseMs,
                QdrantMs: qdrantMs,
                ExactMs: exactMs,
                QuotedTitleMs: quotedTitleMs,
                LocalTitleTokenMs: localTitleTokenMs,
                TitleAnchorRouteMs: titleAnchorRoutePhaseMs,
                SparsePhaseMs: sparsePhaseMs,
                DenseMs: densePhaseMs,
                ProfileMs: profilePhaseMs,
                LinkedMs: linkedPhaseMs,
                FusionMs: fusionMs,
                RerankPhaseMs: rerankPhaseMs,
                SelectionMs: selectionMs),
            Matches: selected.Take(requestedTopK).ToList(),
            DegradedRetrievers: degradedRetrievers.Count == 0
                ? null
                : degradedRetrievers.OrderBy(static retriever => retriever, StringComparer.Ordinal).ToArray(),
            DegradedRetrieverErrors: degradedRetrieverErrors.Count == 0
                ? null
                : degradedRetrieverErrors
                    .OrderBy(static pair => pair.Key, StringComparer.Ordinal)
                    .ToDictionary(static pair => pair.Key, static pair => pair.Value, StringComparer.Ordinal),
            Diagnostics: diagnostics?.Build(selected));

        RetrievalTelemetry.CompleteSearch(searchActivity, response, mode, hasCategoryFilter, hasDocScope);
        RetrievalTelemetry.RecordSearch(response, mode, hasCategoryFilter, hasDocScope, exactMs, sparsePhaseMs + titleAnchorRoutePhaseMs + profilePhaseMs, densePhaseMs, linkedPhaseMs);

        return response;
    }
